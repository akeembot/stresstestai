# Any trading agent can call StressTest AI before executing
import requests

analysis = requests.post('http://localhost:5000/api/stress-test', json={
  "trade_idea": "Buy NVDA before earnings",
  "asset":      "NVDA",
  "direction":  "long",
  "timeframe":  "1 week"
}).json()

if analysis['risk_score'] > 60:
    print("High risk — reviewing scenarios before executing")
